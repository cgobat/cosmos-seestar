COMMAND S50 SET_TIME LITTLE_ENDIAN "Set device time"
  ACCESSOR JsonAccessor
  TEMPLATE '{"id": 1, "method": "pi_set_time", "params": [{"year": 2000, "mon": 1, "day": 1, "hour": 0, "min": 0, "sec": 0, "time_zone": "Etc/UTC"}]}'
  APPEND_ID_PARAMETER METHOD 128 STRING "pi_set_time" "RPC method"
    KEY $.method
  APPEND_PARAMETER YEAR 16 UINT 1970 2170 2020 "Timestamp year"
    KEY $.params[0].year
  APPEND_PARAMETER MONTH 8 UINT 1 12 1 "Timestamp month"
    KEY $.params[0].mon
  APPEND_PARAMETER DAY 8 UINT 1 31 1 "Timestamp day"
    KEY $.params[0].day
  APPEND_PARAMETER HOUR 8 UINT 0 24 0 "Timestamp hour"
    KEY $.params[0].hour
  APPEND_PARAMETER MINUTE 8 UINT 0 60 0 "Timestamp minute"
    KEY $.params[0].min
  APPEND_PARAMETER SECOND 8 UINT 0 60 0 "Timestamp second"
    KEY $.params[0].sec
  APPEND_PARAMETER TIMEZONE 512 STRING "Etc/UTC" "Time zone"
    KEY $.params[0].time_zone
    STATE "UTC" "Etc/UTC"
    STATE "Pacific" "America/Los Angeles"
    STATE "Mountain" "America/Denver"
    STATE "Central" "America/Chicago"
    STATE "Eastern" "America/New York"
